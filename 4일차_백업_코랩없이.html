3️⃣ 에이전트

💻 핵심 코드 (백업 실행기에 붙여넣기)

import numpy as np # 간단한 텍스트 에이전트 (오프라인) class TextAgent: def __init__(self, name, corpus): self.name = name self.corpus = corpus # 문서 리스트 # 단어 집합 만들기 self.vocab = sorted(set(" ".join(corpus).split())) def vectorize(self, text): # 단어를 벡터로 (출현 횟수) words = text.split() return np.array([words.count(w) for w in self.vocab], dtype=float) def cosine_sim(self, a, b): # 코사인 유사도 denom = np.linalg.norm(a) * np.linalg.norm(b) if denom == 0: return 0 return np.dot(a, b) / denom def search(self, query, top_k=2): qv = self.vectorize(query) scores = [] for doc in self.corpus: dv = self.vectorize(doc) scores.append((self.cosine_sim(qv, dv), doc)) scores.sort(reverse=True) return scores[:top_k] def answer(self, query): results = self.search(query) print(f"🤖 [{self.name}] 질문: '{query}'") print(f"📚 가장 관련 있는 문서 (유사도 {results[0][0]*100:.1f}%):") print(f' "{results[0][1]}"') return results[0][1] # 문서 준비 corpus = [ "광화문은 조선시대 경복궁의 정문이다", "경복궁은 1395년에 창건된 조선의 법궁이다", "숭례문은 국보 제1호이다", ] agent = TextAgent("역사도우미", corpus) agent.answer("경복궁 정문은?")
